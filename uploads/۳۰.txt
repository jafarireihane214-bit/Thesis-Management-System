class DatabaseManager:
    """مدیریت تمام عملیات پایگاه داده"""
    
    _instance = None
    _lock = threading.Lock()
    
    def __new__(cls):
        if cls._instance is None:
            with cls._lock:
                if cls._instance is None:
                    cls._instance = super().__new__(cls)
        return cls._instance
    
    def __init__(self):
        if not hasattr(self, 'initialized'):
            self.initialized = True
            self._local = threading.local()
            self._ensure_database()
    
    def _get_connection(self):
        if not hasattr(self._local, 'connection'):
            self._local.connection = sqlite3.connect(DATABASE_FILE, check_same_thread=False)
            self._local.connection.row_factory = sqlite3.Row
        return self._local.connection
    
    def execute(self, query, params=None):
        """اجرای دستور SQL"""
        conn = self._get_connection()
        cursor = conn.cursor()
        try:
            if params:
                cursor.execute(query, params)
            else:
                cursor.execute(query)
            conn.commit()
            return cursor
        except Exception as e:
            conn.rollback()
            raise e
    
    def fetch_all(self, query, params=None):
        cursor = self.execute(query, params)
        return cursor.fetchall()
    
    def fetch_one(self, query, params=None):
        cursor = self.execute(query, params)
        return cursor.fetchone()
    
    def _ensure_database(self):
        """ایجاد جداول پایگاه داده در صورت عدم وجود"""
        queries = [
            """CREATE TABLE IF NOT EXISTS users (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                username TEXT UNIQUE NOT NULL,
                password TEXT NOT NULL,
                email TEXT NOT NULL,
                full_name TEXT,
                role TEXT DEFAULT 'user',
                is_active INTEGER DEFAULT 1,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                last_login TIMESTAMP
            )""",
            """CREATE TABLE IF NOT EXISTS projects (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT NOT NULL,
                description TEXT,
                owner_id INTEGER,
                status TEXT DEFAULT 'Active',
                start_date DATE,
                end_date DATE,
                is_archived INTEGER DEFAULT 0,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY (owner_id) REFERENCES users (id)
            )""",
            """CREATE TABLE IF NOT EXISTS project_members (
                project_id INTEGER,
                user_id INTEGER,
                role TEXT DEFAULT 'member',
                joined_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY (project_id) REFERENCES projects (id),
                FOREIGN KEY (user_id) REFERENCES users (id),
                PRIMARY KEY (project_id, user_id)
            )""",
            """CREATE TABLE IF NOT EXISTS tasks (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                title TEXT NOT NULL,
                description TEXT,
                project_id INTEGER,
                assignee_id INTEGER,
                creator_id INTEGER,
                priority TEXT DEFAULT 'Medium',
                status TEXT DEFAULT 'Todo',
                due_date DATE,
                progress INTEGER DEFAULT 0,
                position INTEGER DEFAULT 0,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY (project_id) REFERENCES projects (id),
                FOREIGN KEY (assignee_id) REFERENCES users (id),
                FOREIGN KEY (creator_id) REFERENCES users (id)
            )""",
            """CREATE TABLE IF NOT EXISTS subtasks (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                task_id INTEGER,
                title TEXT NOT NULL,
                is_completed INTEGER DEFAULT 0,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY (task_id) REFERENCES tasks (id) ON DELETE CASCADE
            )""",
            """CREATE TABLE IF NOT EXISTS comments (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                task_id INTEGER,
                user_id INTEGER,
                content TEXT NOT NULL,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY (task_id) REFERENCES tasks (id) ON DELETE CASCADE,
                FOREIGN KEY (user_id) REFERENCES users (id)
            )""",
            """CREATE TABLE IF NOT EXISTS attachments (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                task_id INTEGER,
                filename TEXT NOT NULL,
                filepath TEXT NOT NULL,
                filesize INTEGER,
                user_id INTEGER,
                uploaded_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY (task_id) REFERENCES tasks (id) ON DELETE CASCADE,
                FOREIGN KEY (user_id) REFERENCES users (id)
            )""",
            """CREATE TABLE IF NOT EXISTS activity_log (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                user_id INTEGER,
                action_type TEXT,
                target_type TEXT,
                target_id INTEGER,
                details TEXT,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY (user_id) REFERENCES users (id)
            )""",
            """CREATE TABLE IF NOT EXISTS notifications (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                user_id INTEGER,
                title TEXT NOT NULL,
                message TEXT NOT NULL,
                is_read INTEGER DEFAULT 0,
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY (user_id) REFERENCES users (id)
            )"""
        ]
        
        for query in queries:
            self.execute(query)
        
        # ایجاد کاربر ادمین پیش‌فرض در صورت عدم وجود
        admin = self.fetch_one("SELECT id FROM users WHERE username = ?", ('admin',))
        if not admin:
            hashed = hashlib.sha256("admin123".encode()).hexdigest()
            self.execute(
                "INSERT INTO users (username, password, email, full_name, role) VALUES (?, ?, ?, ?, ?)",
                ('admin', hashed, 'admin@example.com', 'مدیر سیستم', 'admin')
            )
    
    def backup(self):
        """ایجاد پشتیبان از پایگاه داده"""
        if not os.path.exists(BACKUP_FOLDER):
            os.makedirs(BACKUP_FOLDER)
        
        timestamp = datetime.datetime.now().strftime("%Y%m%d_%H%M%S")
        backup_file = os.path.join(BACKUP_FOLDER, f"backup_{timestamp}.db")
        shutil.copy2(DATABASE_FILE, backup_file)
        return backup_file
    
    def restore(self, backup_file):
        """بازیابی پایگاه داده از فایل پشتیبان"""
        shutil.copy2(backup_file, DATABASE_FILE)
        self._ensure_database()

    id: int

class Project:
    id: int

    description: str
    owner_id: int
    status: str
    start_date: str
    end_date: str
    is_archived: bool
    
    @property
    def progress(self):
        """محاسبه پیشرفت پروژه بر اساس تسک‌ها"""
        db = DatabaseManager()
        tasks = db.fetch_all("SELECT status FROM tasks WHERE project_id = ?", (self.id,))
        if not tasks:
            return 0
        completed = sum(1 for t in tasks if t['status'] == 'انجام شده')
        return int((completed / len(tasks)) * 100)

@dataclass
class Task:
    id: int
    title: str
    description: str
    project_id: int
    assignee_id: Optional[int]
    priority: str
    status: str
    due_date: Optional[str]
    progress: int
    
    @property
    def is_overdue(self):
        if not self.due_date or self.status == 'انجام شده':
            return False
        due = datetime.datetime.strptime(self.due_date, "%Y-%m-%d").date()
        return due < datetime.date.today()

class AuthSystem:
    """مدیریت احراز هویت کاربران و نشست‌ها"""
    
    def __init__(self):
        self.db = DatabaseManager()
        self.current_user: Optional[User] = None
        self.session_token = None
    
    def register(self, username, password, email, full_name):
        """ثبت نام کاربر جدید"""
        hashed = hashlib.sha256(password.encode()).hexdigest()
        try:
            self.db.execute(
                "INSERT INTO users (username, password, email, full_name) VALUES (?, ?, ?, ?)",
                (username, hashed, email, full_name)
            )
            return True, "ثبت نام با موفقیت انجام شد!"
        except sqlite3.IntegrityError:
            return False, "این نام کاربری قبلاً ثبت شده است!"
    
    def login(self, username, password):
        """ورود کاربر به سیستم"""
        hashed = hashlib.sha256(password.encode()).hexdigest()
        user = self.db.fetch_one(
            "SELECT * FROM users WHERE username = ? AND password = ? AND is_active = 1",
            (username, hashed)
        )
        if user:
            self.current_user = User(
                id=user['id'],
                username=user['username'],
                email=user['email'],
                full_name=user['full_name'],
                role=user['role'],
                is_active=bool(user['is_active'])
            )
            self.db.execute(
                "UPDATE users SET last_login = CURRENT_TIMESTAMP WHERE id = ?",
                (self.current_user.id,)
            )
            return True, "ورود موفقیت آمیز بود!"
        return False, "نام کاربری یا رمز عبور اشتباه است یا حساب کاربری غیرفعال شده است!"
    
    def logout(self):
        self.current_user = None
        self.session_token = None
    
    def change_password(self, old_password, new_password):
        if not self.current_user:
            return False, "وارد سیستم نشده‌اید!"
        
        old_hashed = hashlib.sha256(old_password.encode()).hexdigest()
        user = self.db.fetch_one(
            "SELECT id FROM users WHERE id = ? AND password = ?",
            (self.current_user.id, old_hashed)
        )
        if not user:
            return False, "رمز عبور فعلی اشتباه است!"
        
        new_hashed = hashlib.sha256(new_password.encode()).hexdigest()
        self.db.execute(
            "UPDATE users SET password = ? WHERE id = ?",
            (new_hashed, self.current_user.id)
        )
        return True, "رمز عبور با موفقیت تغییر کرد!"

class ActivityLogger:
    """ثبت تمام فعالیت‌های سیستم"""
    
    def __init__(self):
        self.db = DatabaseManager()
        self.queue = queue.Queue()
        self.worker = threading.Thread(target=self._process_logs, daemon=True)
        self.worker.start()
    
    def _process_logs(self):
        while True:
            try:
                log_data = self.queue.get()
                self.db.execute(
                    "INSERT INTO activity_log (user_id, action_type, target_type, target_id, details) VALUES (?, ?, ?, ?, ?)",
                    log_data
                )
            except:
                pass
    
    def log(self, user_id, action_type, target_type, target_id=None, details=None):
        self.queue.put((user_id, action_type, target_type, target_id, details))
    
    def get_activity(self, limit=100, user_id=None, target_type=None):
        query = """
            SELECT a.*, u.username 
            FROM activity_log a
            LEFT JOIN users u ON a.user_id = u.id
            WHERE 1=1
        """
        params = []
        
        if user_id:
            query += " AND a.user_id = ?"
            params.append(user_id)
        if target_type:
            query += " AND a.target_type = ?"
            params.append(target_type)
        
        query += " ORDER BY a.created_at DESC LIMIT ?"
        params.append(limit)
        
        return self.db.fetch_all(query, params)

class ProjectManager:
    """مدیریت عملیات مربوط به پروژه‌ها"""
    
    def __init__(self):
        self.db = DatabaseManager()
        self.logger = ActivityLogger()
    
    def create_project(self, name, description, owner_id, start_date, end_date):
        project_id = self.db.execute(
            """INSERT INTO projects (name, description, owner_id, start_date, end_date)
               VALUES (?, ?, ?, ?, ?)""",
            (name, description, owner_id, start_date, end_date)
        ).lastrowid
        
        # اضافه کردن مالک به عنوان عضو پروژه
        self.db.execute(
            "INSERT INTO project_members (project_id, user_id, role) VALUES (?, ?, ?)",
            (project_id, owner_id, 'owner')
        )
        
        self.logger.log(owner_id, 'create', 'project', project_id, f"ایجاد پروژه: {name}")
        return project_id
    
    def get_user_projects(self, user_id):
        return self.db.fetch_all("""
            SELECT p.*, u.username as owner_name
            FROM projects p
            JOIN project_members pm ON p.id = pm.project_id
            LEFT JOIN users u ON p.owner_id = u.id
            WHERE pm.user_id = ? AND p.is_archived = 0
            ORDER BY p.created_at DESC
        """, (user_id,))
    
    def add_member(self, project_id, user_id, role='member'):
        self.db.execute(
            "INSERT OR IGNORE INTO project_members (project_id, user_id, role) VALUES (?, ?, ?)",
            (project_id, user_id, role)
        )

class TaskManager:
    """مدیریت عملیات مربوط به تسک‌ها"""
    
    def __init__(self):
        self.db = DatabaseManager()
        self.logger = ActivityLogger()
        self.notification_queue = queue.Queue()
    
    def create_task(self, title, description, project_id, assignee_id, creator_id, priority, due_date):
        # دریافت حداکثر موقعیت برای وضعیت
        max_pos = self.db.fetch_one(
            "SELECT MAX(position) as max_pos FROM tasks WHERE project_id = ? AND status = 'انجام نشده'",
            (project_id,)
        )['max_pos'] or -1
        
        task_id = self.db.execute(
            """INSERT INTO tasks (title, description, project_id, assignee_id, creator_id, priority, due_date, position)
               VALUES (?, ?, ?, ?, ?, ?, ?, ?)""",
            (title, description, project_id, assignee_id, creator_id, priority, due_date, max_pos + 1)
        ).lastrowid
        
        self.logger.log(creator_id, 'create', 'task', task_id, f"ایجاد تسک: {title}")
        
        # ایجاد اعلان برای مسئول تسک
        if assignee_id and assignee_id != creator_id:
            self.db.execute(
                "INSERT INTO notifications (user_id, title, message) VALUES (?, ?, ?)",
                (assignee_id, "تسک جدید به شما اختصاص یافت", f"تسک جدید به شما اختصاص یافت: {title}")
            )
        
        return task_id
    
    def update_task_status(self, task_id, new_status, user_id):
        self.db.execute(
            "UPDATE tasks SET status = ?, updated_at = CURRENT_TIMESTAMP WHERE id = ?",
            (new_status, task_id)
        )
        self.logger.log(user_id, 'update', 'task', task_id, f"تغییر وضعیت به {new_status}")
    
    def get_tasks_by_status(self, project_id):
        tasks = {}
        for status in TASK_STATUSES:
            tasks[status] = self.db.fetch_all(
                "SELECT t.*, u.username as assignee_name FROM tasks t "
                "LEFT JOIN users u ON t.assignee_id = u.id "
                "WHERE t.project_id = ? AND t.status = ? "
                "ORDER BY t.position, t.priority DESC",
                (project_id, status)
            )
        return tasks
    
    def move_task(self, task_id, from_status, to_status, position, user_id):
        # به‌روزرسانی وضعیت و موقعیت
        self.db.execute(
            "UPDATE tasks SET status = ?, position = ?, updated_at = CURRENT_TIMESTAMP WHERE id = ?",
            (to_status, position, task_id)
        )
        
        # مرتب‌سازی مجدد تسک‌ها در ستون مبدأ
        self.db.execute(
            "UPDATE tasks SET position = position - 1 WHERE status = ? AND position > ?",
            (from_status, position)
        )
        
        self.logger.log(user_id, 'move', 'task', task_id, f"انتقال از {from_status} به {to_status}")

        card.place(relx=0.5, rely=0.5, anchor="center", width=450, height=600)
        
        tk.Label(card, text="📝 ایجاد حساب کاربری", font=self.fonts['title'],
                bg=COLORS["white"], fg=COLORS["primary"]).pack(pady=20)
        
        form = tk.Frame(card, bg=COLORS["white"])
        form.pack(pady=20, padx=40)
        
        fields = [
            ("نام کاربری", "reg_username"),
            ("نام کامل", "reg_fullname"),
            ("ایمیل", "reg_email"),
            ("رمز عبور", "reg_password"),
            ("تکرار رمز عبور", "reg_confirm")
        ]
        
        self.reg_entries = {}
        for label, key in fields:
            tk.Label(form, text=label, bg=COLORS["white"], font=self.fonts['normal']).pack(anchor="w", pady=(10,5))
            if "password" in key:
                entry = tk.Entry(form, show="•", font=self.fonts['normal'], width=35)
            else:
                entry = tk.Entry(form, font=self.fonts['normal'], width=35)
            entry.pack()
            self.reg_entries[key] = entry
        
        def do_register():
            username = self.reg_entries['reg_username'].get()
            fullname = self.reg_entries['reg_fullname'].get()
            email = self.reg_entries['reg_email'].get()
            password = self.reg_entries['reg_password'].get()
            confirm = self.reg_entries['reg_confirm'].get()
            
            if not all([username, fullname, email, password]):
                messagebox.showerror("خطا", "تمامی فیلدها الزامی هستند!")
                return
            
            if password != confirm:
                messagebox.showerror("خطا", "رمز عبور و تکرار آن مطابقت ندارند!")
                return
            
            success, message = self.auth.register(username, password, email, fullname)
            if success:
                messagebox.showinfo("موفقیت", message)
                self.show_login_frame()
            else:
                messagebox.showerror("خطا", message)
        
        tk.Button(form, text="ثبت نام", command=do_register,
                 bg=COLORS["success"], fg="white", font=self.fonts['bold'],
                 padx=30, pady=10, cursor="hand2").pack(pady=20)
        
        tk.Button(form, text="بازگشت به صفحه ورود", command=self.show_login_frame,
                 bg=COLORS["gray"], fg="white", font=self.fonts['normal'],
                 padx=20, pady=5, cursor="hand2").pack()

    def show_dashboard(self):
        """نمایش داشبورد با آمار"""
        self.clear_content()
        
        # هدر
        header = tk.Frame(self.content_frame, bg=COLORS["white"])
        header.pack(fill="x", padx=20, pady=10)
        tk.Label(header, text="داشبورد", font=self.fonts['title'],
                bg=COLORS["white"]).pack(side="right")
        
        # کارت‌های آماری
        stats_frame = tk.Frame(self.content_frame, bg=COLORS["light"])
        stats_frame.pack(fill="x", padx=20, pady=10)
        
        # دریافت آمار
        user_id = self.auth.current_user.id
        
        my_tasks = self.db.fetch_all(
            "SELECT * FROM tasks WHERE assignee_id = ? AND status != 'انجام شده'",
            (user_id,)
        )
        overdue = self.db.fetch_all(
            "SELECT * FROM tasks WHERE assignee_id = ? AND due_date < date('now') AND status != 'انجام شده'",
            (user_id,)
        )
        projects = self.project_manager.get_user_projects(user_id)
        completed = self.db.fetch_all(
            "SELECT * FROM tasks WHERE assignee_id = ? AND status = 'انجام شده'",
            (user_id,)
        )
        
        stats = [
            ("📋 تسک‌های من", len(my_tasks), COLORS["primary"]),
            ("⚠️ عقب افتاده", len(overdue), COLORS["danger"]),
            ("📁 پروژه‌ها", len(projects), COLORS["success"]),
            ("✅ تکمیل شده", len(completed), COLORS["warning"])
        ]
        
        for title, value, color in stats:
            card = tk.Frame(stats_frame, bg=COLORS["white"], relief="raised", bd=1)
            card.pack(side="right", padx=10, fill="both", expand=True)
            
            tk.Label(card, text=title, font=self.fonts['normal'], bg=COLORS["white"]).pack(pady=10)
            tk.Label(card, text=str(value), font=(FONT_FAMILY, 32, "bold"),
                    fg=color, bg=COLORS["white"]).pack(pady=10)
        
        # نمودار توزیع تسک‌ها
        chart_frame = tk.Frame(self.content_frame, bg=COLORS["white"])
        chart_frame.pack(fill="both", expand=True, padx=20, pady=20)
        
        tk.Label(chart_frame, text="توزیع تسک‌ها", font=self.fonts['header'],
                bg=COLORS["white"]).pack(anchor="w", pady=10)
        
        # نمودار میله‌ای ساده با Canvas
        canvas = tk.Canvas(chart_frame, bg=COLORS["white"], height=300)
        canvas.pack(fill="both", expand=True)
        
        status_counts = {}
        for status in TASK_STATUSES:
            count = self.db.fetch_one(
                "SELECT COUNT(*) as cnt FROM tasks WHERE assignee_id = ? AND status = ?",
                (user_id, status)
            )['cnt']
            status_counts[status] = count
        
        max_count = max(status_counts.values()) if status_counts else 1
        bar_width = 100
        x_start = 100
        
        for i, (status, count) in enumerate(status_counts.items()):
            x = x_start + i * (bar_width + 50)
            bar_height = (count / max_count) * 200 if max_count > 0 else 0
            y = 250 - bar_height
            
            canvas.create_rectangle(x, y, x + bar_width, 250, fill=STATUS_COLORS[status], outline="")
            canvas.create_text(x + bar_width/2, 260, text=status, anchor="n", font=self.fonts['small'])
            canvas.create_text(x + bar_width/2, y - 10, text=str(count), anchor="s", font=self.fonts['normal'])

    def calculate_project_progress(self, project_id):
        """محاسبه درصد پیشرفت پروژه"""
        tasks = self.db.fetch_all("SELECT status FROM tasks WHERE project_id = ?", (project_id,))
        if not tasks:
            return 0
        completed = sum(1 for t in tasks if t['status'] == 'انجام شده')
        return int((completed / len(tasks)) * 100)

                                          value=task['progress'])
            progress_bar.pack(pady=2, padx=5)

        actions = tk.Frame(card, bg=COLORS["white"])
        actions.pack(fill="x", padx=5, pady=5)

        status_order = ["انجام نشده", "در حال انجام", "انجام شده"]
        current_index = status_order.index(current_status)

            tk.Button(actions, text="→", command=lambda: self.move_task(task['id'], current_status,

                       status_order[current_index + 1], project_id),
                     width=3, bg=COLORS["primary"], fg="white", cursor="hand2",

                 font=self.fonts['normal']).pack(side="right", padx=2)

            menu.add_separator()

    def show_comments(self, task_id):
        """نمایش دیالوگ نظرات تسک"""
        dialog = tk.Toplevel(self.root)
        dialog.title("نظرات تسک")
        dialog.geometry("600x500")
        
        # لیست نظرات
        comments_frame = tk.Frame(dialog)
        comments_frame.pack(fill="both", expand=True, padx=10, pady=10)
        
        comments_list = tk.Text(comments_frame, wrap="word", height=15, font=self.fonts['normal'])
        comments_list.pack(fill="both", expand=True)
        
        scrollbar = ttk.Scrollbar(comments_list)
        scrollbar.pack(side="right", fill="y")
        comments_list.config(yscrollcommand=scrollbar.set)
        scrollbar.config(command=comments_list.yview)
        
        # بارگذاری نظرات
        comments = self.db.fetch_all(
            "SELECT c.*, u.username FROM comments c JOIN users u ON c.user_id = u.id WHERE c.task_id = ? ORDER BY c.created_at DESC",
            (task_id,)
        )
        
        for comment in comments:
            comments_list.insert("end", f"{comment['username']} ({comment['created_at']}):\n")
            comments_list.insert("end", f"{comment['content']}\n")
            comments_list.insert("end", "-" * 50 + "\n\n")
        
        comments_list.config(state="disabled")
        
        # افزودن نظر جدید
        add_frame = tk.Frame(dialog)
        add_frame.pack(fill="x", padx=10, pady=10)
        
        tk.Label(add_frame, text="نظر جدید:", font=self.fonts['normal']).pack(anchor="w")
        comment_entry = tk.Text(add_frame, height=5, width=70, font=self.fonts['normal'])
        comment_entry.pack(fill="x", pady=5)
        
        def add_comment():
            content = comment_entry.get("1.0", "end-1c").strip()
            if content:
                self.db.execute(
                    "INSERT INTO comments (task_id, user_id, content) VALUES (?, ?, ?)",
                    (task_id, self.auth.current_user.id, content)
                )
                self.logger.log(self.auth.current_user.id, 'comment', 'task', task_id, f"افزودن نظر: {content[:50]}")
                dialog.destroy()
                self.show_comments(task_id)
                self.update_status("نظر با موفقیت اضافه شد")
        
        tk.Button(add_frame, text="افزودن نظر", command=add_comment,
                 bg=COLORS["primary"], fg="white", cursor="hand2",
                 font=self.fonts['normal']).pack(pady=5)
    
    def show_search(self):
        """رابط جستجوی پیشرفته"""
        self.clear_content()
        
        header = tk.Frame(self.content_frame, bg=COLORS["white"])
        header.pack(fill="x", padx=20, pady=10)
        tk.Label(header, text="جستجو", font=self.fonts['title'],
                bg=COLORS["white"]).pack(side="right")
        
        # فرم جستجو
        search_frame = tk.Frame(self.content_frame, bg=COLORS["white"])
        search_frame.pack(fill="x", padx=20, pady=10)
        
        # عبارت جستجو
        tk.Label(search_frame, text="عبارت جستجو:", bg=COLORS["white"],
                font=self.fonts['normal']).grid(row=0, column=0, sticky="w", pady=5)
        search_term = tk.Entry(search_frame, width=40, font=self.fonts['normal'])
        search_term.grid(row=0, column=1, pady=5, padx=10)
        
        # فیلترها
        tk.Label(search_frame, text="نوع:", bg=COLORS["white"],
                font=self.fonts['normal']).grid(row=1, column=0, sticky="w", pady=5)
        search_type = ttk.Combobox(search_frame, values=["تسک‌ها", "پروژه‌ها", "کاربران"], width=20)
        search_type.set("تسک‌ها")
        search_type.grid(row=1, column=1, pady=5, padx=10)
        
        tk.Label(search_frame, text="وضعیت:", bg=COLORS["white"],
                font=self.fonts['normal']).grid(row=2, column=0, sticky="w", pady=5)
        status_filter = ttk.Combobox(search_frame, values=["همه"] + TASK_STATUSES, width=20)
        status_filter.set("همه")
        status_filter.grid(row=2, column=1, pady=5, padx=10)
        
        tk.Label(search_frame, text="اولویت:", bg=COLORS["white"],
                font=self.fonts['normal']).grid(row=3, column=0, sticky="w", pady=5)
        priority_filter = ttk.Combobox(search_frame, values=["همه"] + PRIORITIES, width=20)
        priority_filter.set("همه")
        priority_filter.grid(row=3, column=1, pady=5, padx=10)
        
        results_frame = tk.Frame(self.content_frame, bg=COLORS["light"])
        results_frame.pack(fill="both", expand=True, padx=20, pady=10)
        
        def do_search():
            # پاک کردن نتایج قبلی
            for widget in results_frame.winfo_children():
                widget.destroy()
            
            term = search_term.get().lower()
            s_type = search_type.get()
            status = status_filter.get()
            priority = priority_filter.get()
            
            if s_type == "تسک‌ها":
                query = """
                    SELECT t.*, p.name as project_name, u.username as assignee_name
                    FROM tasks t
                    LEFT JOIN projects p ON t.project_id = p.id
                    LEFT JOIN users u ON t.assignee_id = u.id
                    WHERE (t.title LIKE ? OR t.description LIKE ?)
                """
                params = [f"%{term}%", f"%{term}%"]
                
                if status != "همه":
                    query += " AND t.status = ?"
                    params.append(status)
                if priority != "همه":
                    query += " AND t.priority = ?"
                    params.append(priority)
                
                results = self.db.fetch_all(query, params)
                
                for result in results:
                    self.create_task_card(results_frame, result)
            
            elif s_type == "پروژه‌ها":
                query = "SELECT * FROM projects WHERE name LIKE ? OR description LIKE ?"
                params = [f"%{term}%", f"%{term}%"]
                results = self.db.fetch_all(query, params)
                
                for result in results:
                    self.create_project_card(results_frame, result)
            
            elif s_type == "کاربران":
                query = "SELECT * FROM users WHERE username LIKE ? OR full_name LIKE ? OR email LIKE ?"
                params = [f"%{term}%", f"%{term}%", f"%{term}%"]
                results = self.db.fetch_all(query, params)
                
                for result in results:
                    self.create_user_card(results_frame, result)
        
        tk.Button(search_frame, text="جستجو", command=do_search,
                 bg=COLORS["primary"], fg="white", font=self.fonts['bold'],
                 padx=20, pady=5, cursor="hand2").grid(row=4, column=0, columnspan=2, pady=20)

        report_type = ttk.Combobox(report_frame, values=["تسک‌ها بر اساس پروژه", "تسک‌ها بر اساس کاربر", "نرخ تکمیل", "تسک‌های عقب افتاده"], width=25)
        report_type.set("تسک‌ها بر اساس پروژه")
        report_type.pack(side="right", padx=5)
        
        def generate_report():
            r_type = report_type.get()
            results_frame = tk.Frame(self.content_frame, bg=COLORS["light"])
            results_frame.pack(fill="both", expand=True, padx=20, pady=10)
            
            # پاک کردن نتایج قبلی
            for widget in results_frame.winfo_children():
                widget.destroy()
            
            if r_type == "تسک‌ها بر اساس پروژه":
                data = self.db.fetch_all("""
                    SELECT p.name, COUNT(t.id) as task_count
                    FROM projects p
                    LEFT JOIN tasks t ON p.id = t.project_id
                    GROUP BY p.id
                    ORDER BY task_count DESC
                """)
                
                # ایجاد نمودار میله‌ای
                canvas = tk.Canvas(results_frame, bg=COLORS["white"], height=400)
                canvas.pack(fill="both", expand=True)
                
                max_count = max([d['task_count'] for d in data]) if data else 1
                bar_width = 50
                x_start = 100
                
                for i, d in enumerate(data):
                    x = x_start + i * (bar_width + 30)
                    bar_height = (d['task_count'] / max_count) * 300 if max_count > 0 else 0
                    y = 350 - bar_height
                    
                    canvas.create_rectangle(x, y, x + bar_width, 350, fill=COLORS["primary"], outline="")
                    canvas.create_text(x + bar_width/2, 370, text=d['name'][:15], anchor="n", angle=45, font=self.fonts['small'])
                    canvas.create_text(x + bar_width/2, y - 10, text=str(d['task_count']), anchor="s", font=self.fonts['normal'])
            
            elif r_type == "تسک‌ها بر اساس کاربر":
                data = self.db.fetch_all("""
                    SELECT u.username, COUNT(t.id) as task_count
                    FROM users u
                    LEFT JOIN tasks t ON u.id = t.assignee_id
                    GROUP BY u.id
                    ORDER BY task_count DESC
                """)
                
                results_text = tk.Text(results_frame, wrap="word", height=20, font=self.fonts['normal'])
                results_text.pack(fill="both", expand=True)
                
                results_text.insert("end", "گزارش تسک‌ها بر اساس کاربر\n")
                results_text.insert("end", "=" * 50 + "\n\n")
                for d in data:
                    results_text.insert("end", f"{d['username']}: {d['task_count']} تسک\n")
                
                results_text.config(state="disabled")
            
            elif r_type == "نرخ تکمیل":
                data = self.db.fetch_all("""
                    SELECT status, COUNT(*) as count
                    FROM tasks
                    GROUP BY status
                """)
                
                # شبیه‌سازی نمودار دایره‌ای
                canvas = tk.Canvas(results_frame, bg=COLORS["white"], height=400)
                canvas.pack(fill="both", expand=True)
                
                total = sum(d['count'] for d in data)
                start_angle = 0
                colors = [COLORS["success"], COLORS["primary"], COLORS["gray"], COLORS["danger"]]
                
                for i, d in enumerate(data):
                    extent = (d['count'] / total) * 360 if total > 0 else 0
                    canvas.create_arc(150, 50, 450, 350, start=start_angle, extent=extent,
                                     fill=colors[i % len(colors)], outline="white")
                    start_angle += extent
                
                # راهنما
                legend_y = 50
                for i, d in enumerate(data):
                    canvas.create_rectangle(500, legend_y, 520, legend_y + 20, fill=colors[i % len(colors)], outline="")
                    canvas.create_text(530, legend_y + 10, text=f"{d['status']}: {d['count']} ({d['count']/total*100:.1f}%)", anchor="w", font=self.fonts['normal'])
                    legend_y += 30
            
            elif r_type == "تسک‌های عقب افتاده":
                data = self.db.fetch_all("""
                    SELECT t.title, p.name as project_name, u.username as assignee_name, t.due_date
                    FROM tasks t
                    LEFT JOIN projects p ON t.project_id = p.id
                    LEFT JOIN users u ON t.assignee_id = u.id
                    WHERE t.due_date < date('now') AND t.status != 'انجام شده'
                    ORDER BY t.due_date
                """)
                
                results_text = tk.Text(results_frame, wrap="word", height=20, font=self.fonts['normal'])
                results_text.pack(fill="both", expand=True)
                
                results_text.insert("end", "گزارش تسک‌های عقب افتاده\n")
                results_text.insert("end", "=" * 50 + "\n\n")
                for d in data:
                    results_text.insert("end", f"تسک: {d['title']}\n")
                    results_text.insert("end", f"پروژه: {d['project_name']}\n")
                    results_text.insert("end", f"مسئول: {d['assignee_name']}\n")
                    results_text.insert("end", f"تاریخ سررسید: {d['due_date']}\n")
                    results_text.insert("end", "-" * 30 + "\n")
                
                results_text.config(state="disabled")
        
        tk.Button(report_frame, text="ایجاد گزارش", command=generate_report,
                 bg=COLORS["primary"], fg="white", font=self.fonts['bold'],
                 padx=20, pady=5, cursor="hand2").pack(side="right", padx=10)
        
        # دکمه‌های خروجی
        export_frame = tk.Frame(self.content_frame, bg=COLORS["white"])

        """رابط مدیریت کاربران (فقط ادمین)"""
        if not self.auth.current_user.is_admin:
            messagebox.showerror("دسترسی غیرمجاز", "دسترسی ادمین الزامی است")
            return
        
        self.clear_content()
        
        header = tk.Frame(self.content_frame, bg=COLORS["white"])
        header.pack(fill="x", padx=20, pady=10)
        tk.Label(header, text="مدیریت کاربران", font=self.fonts['title'],
                bg=COLORS["white"]).pack(side="right")
        
        # لیست کاربران
        users_frame = tk.Frame(self.content_frame, bg=COLORS["light"])
        users_frame.pack(fill="both", expand=True, padx=20, pady=10)
        
        # نمای درختی
        columns = ("ID", "نام کاربری", "نام کامل", "ایمیل", "نقش", "وضعیت")
        tree = ttk.Treeview(users_frame, columns=columns, show="headings")
        
        for col in columns:
            tree.heading(col, text=col)
            tree.column(col, width=150)
        
        # بارگذاری کاربران
        users = self.db.fetch_all("SELECT * FROM users ORDER BY id")
        for user in users:
            status = "فعال" if user['is_active'] else "غیرفعال"
            tree.insert("", "end", values=(user['id'], user['username'], user['full_name'],
                                          user['email'], user['role'], status))
        
        tree.pack(fill="both", expand=True)
        
        # دکمه‌های عملیات
        actions = tk.Frame(self.content_frame, bg=COLORS["white"])
        actions.pack(fill="x", padx=20, pady=10)
        
        def edit_user():
            selected = tree.selection()
            if not selected:
                messagebox.showwarning("اخطار", "لطفاً یک کاربر را انتخاب کنید")
                return
            
            user_id = tree.item(selected[0])['values'][0]
            self.edit_user_dialog(user_id)
        
        def toggle_user_status():
            selected = tree.selection()
            if not selected:
                messagebox.showwarning("اخطار", "لطفاً یک کاربر را انتخاب کنید")
                return
            
            user_id = tree.item(selected[0])['values'][0]
            user = self.db.fetch_one("SELECT is_active FROM users WHERE id = ?", (user_id,))
            new_status = 0 if user['is_active'] else 1
            self.db.execute("UPDATE users SET is_active = ? WHERE id = ?", (new_status, user_id))
            self.show_users()
            self.update_status("وضعیت کاربر تغییر کرد")
        
        def delete_user():
            selected = tree.selection()
            if not selected:
                messagebox.showwarning("اخطار", "لطفاً یک کاربر را انتخاب کنید")
                return
            
            user_id = tree.item(selected[0])['values'][0]
            if user_id == self.auth.current_user.id:
                messagebox.showerror("خطا", "نمی‌توانید حساب کاربری خود را حذف کنید")
                return
            
            if messagebox.askyesno("تأیید", "آیا این کاربر حذف شود؟"):
                self.db.execute("DELETE FROM users WHERE id = ?", (user_id,))
                self.show_users()
                self.update_status("کاربر حذف شد")
        
        tk.Button(actions, text="ویرایش کاربر", command=edit_user,
                 bg=COLORS["warning"], fg="white", cursor="hand2",
                 font=self.fonts['normal']).pack(side="right", padx=5)
        
        tk.Button(actions, text="تغییر وضعیت", command=toggle_user_status,
                 bg=COLORS["primary"], fg="white", cursor="hand2",
                 font=self.fonts['normal']).pack(side="right", padx=5)
        
        tk.Button(actions, text="حذف کاربر", command=delete_user,
                 bg=COLORS["danger"], fg="white", cursor="hand2",
                 font=self.fonts['normal']).pack(side="right", padx=5)

                 bg=COLORS["primary"], fg="white", padx=20, pady=5,
                 cursor="hand2", font=self.fonts['bold']).grid(row=6, column=0, columnspan=2, pady=20)
    
    def delete_task(self, task_id):
        """حذف تسک"""
        if messagebox.askyesno("تأیید", "آیا این تسک حذف شود؟"):
            task = self.db.fetch_one("SELECT project_id FROM tasks WHERE id = ?", (task_id,))
            self.db.execute("DELETE FROM tasks WHERE id = ?", (task_id,))
            self.logger.log(self.auth.current_user.id, 'delete', 'task', task_id, "حذف تسک")
            self.show_kanban_board(task['project_id'])
            self.update_status("تسک با موفقیت حذف شد")
    
    def backup_database(self):
        """ایجاد پشتیبان از پایگاه داده"""
        if not os.path.exists(BACKUP_FOLDER):
            os.makedirs(BACKUP_FOLDER)
        
        backup_file = self.db.backup()
        messagebox.showinfo("پشتیبان", f"پشتیبان با موفقیت ایجاد شد: {backup_file}")
        self.update_status("پشتیبان از دیتابیس ایجاد شد")
    
    def is_overdue(self, due_date):
        """بررسی عقب افتادگی تسک"""
        if not due_date:
            return False
        due = datetime.datetime.strptime(due_date, "%Y-%m-%d").date()
        return due < datetime.date.today()
    
    def update_status(self, message):
        """به‌روزرسانی پیام نوار وضعیت"""
        self.status_label.config(text=message)
        self.root.after(3000, lambda: self.status_label.config(text="آماده"))
    
    def clear_content(self):
        """پاک کردن منطقه محتوای اصلی"""
        for widget in self.content_frame.winfo_children():
            if widget not in [self.statusbar]:
                widget.destroy()
    
    def logout(self):
        """خروج کاربر از سیستم"""
        self.auth.logout()
        self.show_login_frame()
        self.update_status("خروج از سیستم انجام شد")

def main():
    """نقطه ورود اصلی برنامه"""
    # ایجاد پوشه‌های مورد نیاز
    for folder in [ATTACHMENTS_FOLDER, BACKUP_FOLDER, EXPORT_FOLDER]:

    # مقداردهی اولیه پایگاه داده
    db = DatabaseManager()

    root = tk.Tk()
    
    # تنظیم فونت پیش‌فرض
    try:
        root.option_add('*Font', f'{FONT_FAMILY} {FONT_SIZE_NORMAL}')
    except:
        pass
    
    app = ProjectManagementApp(root)
    
    # تنظیم آیکون پنجره (در صورت وجود)
    try:
        root.iconbitmap(default='icon.ico')
    except:
        pass
    
    # وسط‌چین کردن پنجره
    root.update_idletasks()
    width = root.winfo_width()
    height = root.winfo_height()
    x = (root.winfo_screenwidth() // 2) - (width // 2)
    y = (root.winfo_screenheight() // 2) - (height // 2)
    root.geometry(f'{width}x{height}+{x}+{y}')
    
    # شروع حلقه اصلی
    root.mainloop()

if __name__ == "__main__":
